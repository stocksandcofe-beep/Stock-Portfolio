function renderHoldings() { Papa.parse(HOLD_CSV, { download: true, header: true, skipEmptyLines: true, complete: function(res) { const tbody = document.getElementById('holdings-table-body'); tbody.innerHTML = ''; res.data.forEach(row => { // 1. IDENTIFY COLUMNS const ticker = findVal(row, ['Ticker', 'Symbol']) || '???'; const shares = parseNum(findVal(row, ['Shares', 'Qty'])); if (shares === 0) return; const currency = (findVal(row, ['Currency']) || 'USD').toUpperCase(); // FX Rate Logic: Performance CSV stores 1 GBP = X Foreign Units (e.g., 1.28 for USD) // To get GBP value, we divide the foreign amount by this rate. const rate = (currency === 'GBP') ? 1 : (currency === 'EUR' ? rates.EUR : rates.USD); const symbol = (currency === 'GBP') ? '£' : (currency === 'EUR' ? '€' : '$'); // 2. MARKET VALUATION (CURRENT) const currentPriceLocal = parseNum(findVal(row, ['Price', 'Current Price'])); const currentValLocal = shares * currentPriceLocal; const currentValGBP = currentValLocal / rate; // 3. COST BASIS (FROM YOUR NEW COLUMN) const totalCostLocal = parseNum(findVal(row, ['Total Cost', 'Cost'])); const costGBP = totalCostLocal / rate; // 4. PERFORMANCE METRICS const gbpProfit = currentValGBP - costGBP; const returnPct = costGBP !== 0 ? (gbpProfit / costGBP) * 100 : 0; // Weighting relative to the whole portfolio const weight = totalPortfolioValue > 0 ? (currentValGBP / totalPortfolioValue) * 100 : 0; tbody.innerHTML += `
${findVal(row, ['Company', 'Name']) || ticker}
${ticker}
${shares} ${symbol}${currentPriceLocal.toFixed(2)} £${currentValGBP.toLocaleString(undefined,{minimumFractionDigits:2, maximumFractionDigits:2})} £${gbpProfit.toLocaleString(undefined,{minimumFractionDigits:2})} ${returnPct.toFixed(2)}%
${weight.toFixed(1)}%
`; }); } }); }
